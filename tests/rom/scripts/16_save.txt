# Saving: settings and campaign progress survive a reboot, and wiping really wipes.
symbols build/moku.sym

write32 $g_moku_test+72 5

write32 $g_moku_test+76 1

write32 $g_moku_test+72 7
wait32 $g_moku_test+72 0 20
frames 6
write32 $g_moku_test+76 1
write32 $g_moku_test+72 13
wait32 $g_moku_test+72 0 20
frames 6

# clear mission 1 for real: load it and place the three marked stones
write32 $g_moku_test+76 1

assert8 $g_moku_test+10 0
assert8 $g_moku_test+11 0
