# A sandbox game against each opponent, played to the end on 7x7, plus hot seat.
# TCMD_START_SANDBOX arg: size index (0=7,1=9,2=13,3=19) | player_black<<4 | level<<8 |
#                         handicap<<12 | rules<<16 | komi_x2<<20 | hotseat<<28
symbols build/moku.sym
wait8 $g_moku_test+8 1 120
frames 20
write32 $g_moku_test+72 8
wait32 $g_moku_test+72 0 20
frames 6

# size 0 (7x7) | black | level 0 | komi 15 -> 0 | 0x10 | 0 | 15<<20

write32 $g_moku_test+72 3
wait32 $g_moku_test+72 0 30
